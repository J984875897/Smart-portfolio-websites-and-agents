  12 def strict_date(value):
  13     if not isinstance(value, str):
  14         return None
  15     try:
  16         parsed = datetime.strptime(value, '%Y-%m-%d').date()
  17         return parsed if parsed.isoformat() == value else None
  18     except ValueError:
  19         return None

  22 def finite_number(value):
  23     if isinstance(value, bool):
  24         return None
  25     try:
  26         result = float(value)
  27         return result if math.isfinite(result) else None

  32 def build_local_returns(bundle, *, adjusted_price_assets=()):
  33     """One result per observation, including failures; evidence joins by asset/date."""
  34     rows = bundle.observations.to_dict('records')
  35     confirmed = set(adjusted_price_assets)
  36     assets = {}
  37     for asset in bundle.assets.to_dict('records'):
  38         assets.setdefault(asset.get('asset_id'), []).append(asset)
  39     groups = {}
  40     parsed = [strict_date(r.get('date')) for r in rows]
  41     for i, (row, date) in enumerate(zip(rows, parsed)):
  42         if date:
  43             groups.setdefault((row.get('asset_id'), date.year, date.month), []).append(i)
  44     output = []
  45     for row, date in zip(rows, parsed):
  46         out = {x: row.get(x, '') for x in LOCAL_COLUMNS}
  47         out.update(local_return=float('nan'), start_price=float('nan'), end_price=float('nan'), start_date='', end_date=row.get('date', ''), status='blocked', reason='')
  48         reason = ''
  49         aid = row.get('asset_id')
  50         currency = row.get('currency')
  51         number = finite_number(row.get('value'))
  52         metadata = assets.get(aid, [])
  53         if date is None: reason = 'invalid_date'
  54         elif len(groups[(aid, date.year, date.month)]) > 1: reason = 'ambiguous_asset_month'
  55         elif currency not in CURRENCIES: reason = 'invalid_currency'
  56         elif len(metadata) != 1: reason = 'unknown_asset' if not metadata else 'ambiguous_asset_metadata'
  57         elif metadata[0].get('currency') != currency: reason = 'currency_conflict'
  58         elif row.get('frequency') not in FREQUENCIES: reason = 'invalid_frequency'
  59         elif row.get('frequency') != 'monthly' or date.day != calendar.monthrange(date.year, date.month)[1]: reason = 'alignment_required'
  60         elif number is None: reason = 'invalid_value'
  61         elif row.get('value_kind') in {'price', 'price_return'}: reason = 'not_total_return'
  62         elif row.get('value_kind') not in {'total_return', 'adjusted_price'}: reason = 'invalid_value_kind'
  63         elif row.get('value_kind') == 'total_return' and number < -1: reason = 'invalid_value'
  64         elif row.get('value_kind') == 'adjusted_price' and number <= 0: reason = 'invalid_value'
  65         elif row.get('value_kind') == 'adjusted_price' and aid not in confirmed: reason = 'unconfirmed_adjusted_price'
  66         if date:
  67             start = date.replace(day=1) - timedelta(days=1)
  68             out['start_date'] = start.isoformat()
  69         if not reason and row['value_kind'] == 'total_return':
  70             out['local_return'] = number
  71         elif not reason:
  72             out['end_price'] = number
  73             indices = groups.get((aid, start.year, start.month), [])
  74             if not indices:
  75                 earlier = any(d and d < date and r.get('asset_id') == aid for r, d in zip(rows, parsed))
  76                 reason = 'missing_previous_month' if earlier else 'first_period'
  77             elif len(indices) != 1:
  78                 reason = 'invalid_previous_month'
  79             else:
  80                 previous = rows[indices[0]]
  81                 p = finite_number(previous.get('value'))
  82                 if parsed[indices[0]] != start or previous.get('frequency') != 'monthly' or previous.get('value_kind') != 'adjusted_price' or previous.get('currency') != currency or p is None or p <= 0:
  83                     reason = 'invalid_previous_month'
  84                 else:
  85                     result = number / p - 1
  86                     if not math.isfinite(result): reason = 'invalid_value'
  87                     else:
  88                         out.update(local_return=result, start_price=p)
  89                         out.update({'start_'+x: previous.get(x, '') for x in SOURCE_FIELDS})
  90         out.update(status='blocked' if reason else 'valid', reason=reason or 'ok')
  91         output.append(out)
  92     return pd.DataFrame(output, columns=LOCAL_COLUMNS)

  25 def validate_fx(frame):
  26     """Reject every record of a conflicting/invalid key; preserve all duplicate evidence."""
  27     if not frame.columns.is_unique:
  28         raise ValueError('FX columns must be unique')
  29     required = ['date', 'currency', 'hkd_per_unit', 'source_name']
  30     payload = frame.attrs.get('raw_bytes', frame.to_csv(index=False).encode())
  31     digest = hashlib.sha256(payload).hexdigest()
  32     groups = {}
  33     issues = []
  34     def issue(row, code):
  35         issues.append({**{x: row.get(x, '') for x in ISSUE_COLUMNS}, 'code': code, 'severity': 'blocked', 'message': code})
  36     missing = [x for x in required if x not in frame.columns]
  37     if missing:
  38         issue({}, 'missing_columns:' + ','.join(missing))
  39         return FxValidation(pd.DataFrame(columns=FX_COLUMNS), pd.DataFrame(issues, columns=ISSUE_COLUMNS))
  40     source_rows = frame.attrs.get('source_rows', list(range(2, len(frame)+2)))
  41     for i, original in enumerate(frame.to_dict('records')):
  42         row = dict(original)
  43         for field, fallback in [('source_file', frame.attrs.get('source_file', 'dataframe.csv')), ('source_sheet', frame.attrs.get('source_sheet', '')), ('source_row', source_rows[i]), ('source_hash', digest)]:
  44             row.setdefault(field, fallback)
  45         number = finite_number(row.get('hkd_per_unit'))
  46         errors = []
  47         if strict_date(row.get('date')) is None: errors.append('invalid_date')
  48         if row.get('currency') not in CURRENCIES: errors.append('invalid_currency')
  49         if number is None or number <= 0: errors.append('invalid_quote')
  50         source = row.get('source_name')
  51         if not isinstance(source, str) or not source.strip(): errors.append('missing_source_name')
  52         row['_number'] = number
  53         row['_errors'] = errors
  54         # String keys also retain malformed dates without making them valid.
  55         groups.setdefault((str(row.get('date')), str(row.get('currency'))), []).append(row)
  56     values = []
  57     for group in groups.values():
  58         invalid = any(r['_errors'] for r in group)
  59         conflict = len({r['_number'] for r in group if r['_number'] is not None}) > 1
  60         if invalid or conflict:
  61             for row in group:
  62                 issue(row, ';'.join(row['_errors']) or ('conflicting_quote' if conflict else 'invalid_key_record'))
  63             continue
  64         first = group[0]
  65         evidence = [{x: r.get(x, '') for x in ['source_name', *SOURCE_FIELDS]} for r in group]
  66         values.append({**{x: first.get(x, '') for x in FX_COLUMNS}, 'hkd_per_unit': first['_number'], 'sources': evidence})
  67     return FxValidation(pd.DataFrame(values, columns=FX_COLUMNS), pd.DataFrame(issues, columns=ISSUE_COLUMNS))

  70 def reconcile_hkd(bundle, fx_frame, *, adjusted_price_assets=()):
  71     local = build_local_returns(bundle, adjusted_price_assets=adjusted_price_assets)
  72     validated = validate_fx(fx_frame)
  73     quotes = {(r['date'], r['currency']): r for r in validated.values.to_dict('records')}
  74     rows = []
  75     for original in local.to_dict('records'):
  76         row = {x: original.get(x, '') for x in RECONCILIATION_COLUMNS}
  77         row.update({x: float('nan') for x in ['fx_start', 'fx_end', 'fx_return', 'cross_term', 'hkd_return']})
  78         row.update(fx_start_sources=[], fx_end_sources=[])
  79         if original['status'] != 'valid':
  80             rows.append(row)
  81             continue
  82         if row['currency'] == 'HKD':
  83             row.update(fx_start=1., fx_end=1., fx_return=0., cross_term=0., hkd_return=row['local_return'])
  84         else:
  85             absent = []
  86             for side in ['start', 'end']:
  87                 quote = quotes.get((row[side+'_date'], row['currency']))
  88                 if quote is None:
  89                     absent.append(side)
  90                 else:
  91                     row['fx_'+side] = quote['hkd_per_unit']
  92                     row['fx_'+side+'_sources'] = quote['sources']
  93                     row.update({'fx_'+side+'_'+x: quote[x] for x in ['source_name', *SOURCE_FIELDS]})
  94             if absent:
  95                 row.update(status='blocked', reason='missing_fx_'+'_and_'.join(absent))
  96             else:
  97                 fx_return = row['fx_end']/row['fx_start']-1
  98                 cross = row['local_return']*fx_return
  99                 result = (1+row['local_return'])*row['fx_end']/row['fx_start']-1
 100                 if all(math.isfinite(x) for x in [fx_return, cross, result]):
 101                     row.update(fx_return=fx_return, cross_term=cross, hkd_return=result)
 102                 else:
 103                     row.update(status='blocked', reason='nonfinite_hkd_result')
 104         rows.append(row)
 105     return HkdResult(local, pd.DataFrame(rows, columns=RECONCILIATION_COLUMNS), validated.issues)
